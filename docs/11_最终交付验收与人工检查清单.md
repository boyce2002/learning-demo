## 五、无 Git 时的清洁 ZIP 命令

在项目根目录执行。命令把源码复制到临时目录，显式排除依赖、构建物、缓存、运行数据、密钥和 `delivery/`，不会删除项目源码：

```powershell
$projectRoot = (Resolve-Path '.').Path
$stageRoot = Join-Path $env:TEMP 'SparkLab_v0.2_source_stage'
$deliveryRoot = Join-Path $projectRoot 'delivery'
$zipPath = Join-Path $deliveryRoot 'SparkLab_v0.2_source.zip'

if (Test-Path -LiteralPath $stageRoot) {
  Remove-Item -LiteralPath $stageRoot -Recurse -Force
}
New-Item -ItemType Directory -Path $stageRoot | Out-Null
New-Item -ItemType Directory -Path $deliveryRoot -Force | Out-Null

robocopy $projectRoot $stageRoot /E `
  /XD .git .venv venv node_modules dist .pytest-tmp .pytest_cache __pycache__ .step5-tmp delivery `
  /XF .env *.sqlite3 *.sqlite3-shm *.sqlite3-wal *.pyc *.pyo *.log *.tsbuildinfo *.zip
if ($LASTEXITCODE -gt 7) { throw "robocopy failed: $LASTEXITCODE" }

if (Test-Path -LiteralPath $zipPath) {
  Remove-Item -LiteralPath $zipPath -Force
}
Compress-Archive -Path (Join-Path $stageRoot '*') -DestinationPath $zipPath -CompressionLevel Optimal
Get-FileHash -Algorithm SHA256 -LiteralPath $zipPath
```

## 六、ZIP 解压复核

```powershell
$verifyRoot = Join-Path $env:TEMP 'SparkLab_v0.2_source_verify'
if (Test-Path -LiteralPath $verifyRoot) {
  Remove-Item -LiteralPath $verifyRoot -Recurse -Force
}
Expand-Archive -LiteralPath $zipPath -DestinationPath $verifyRoot

$required = @(
  'README.md', 'AGENTS.md',
  'backend\requirements.txt', 'backend\app\services.py', 'backend\tests\test_api_flow.py',
  'frontend\package.json', 'frontend\package-lock.json', 'frontend\src\App.tsx',
  'docs\design-rationale.md', 'docs\demo-script.md',
  'docs\10_正式演示分镜与解说词.md', 'docs\11_最终交付验收与人工检查清单.md'
)
$missing = $required | Where-Object { -not (Test-Path -LiteralPath (Join-Path $verifyRoot $_)) }
if ($missing) { throw "Missing required files: $($missing -join ', ')" }

$forbidden = Get-ChildItem -LiteralPath $verifyRoot -Recurse -Force | Where-Object {
  $_.Name -in @('.env', '.venv', 'venv', 'node_modules', 'dist', '.pytest-tmp', '.pytest_cache', '__pycache__', '.step5-tmp') -or
  $_.Name -match '\.(sqlite3|sqlite3-shm|sqlite3-wal|pyc|pyo|log|tsbuildinfo)$'
}
if ($forbidden) { throw "Forbidden entries found in ZIP" }

Get-ChildItem -LiteralPath $verifyRoot -Recurse -File | Measure-Object
```
